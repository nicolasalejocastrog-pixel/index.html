function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); var sheetLlegadas = ss.getSheetByName("Llegadas") || ss.insertSheet("Llegadas"); if (sheetLlegadas.getLastRow() === 0) { sheetLlegadas.appendRow(["Nombre y Apellido", "Sacramento / Motivo", "Día / Fecha", "Hora"]); } var fechaActual = new Date(); var timeZone = ss.getSpreadsheetTimeZone(); var diaFecha = Utilities.formatDate(fechaActual, timeZone, "dd/MM/yyyy"); var horaExacta = Utilities.formatDate(fechaActual, timeZone, "HH:mm:ss"); sheetLlegadas.appendRow([data.nombreCompleto, data.motivo, diaFecha, horaExacta]); return responseJSON({ status: "success", nombre: data.nombreCompleto, motivo: data.motivo, fecha: diaFecha, hora: horaExacta }); } catch (error) { return responseJSON({ status: "error", message: error.toString() }); } finally { lock.releaseLock(); } } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
